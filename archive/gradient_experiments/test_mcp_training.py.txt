"""Nonidentity MCP edits stay real Parameters and can continue learning."""
import asyncio
import importlib.util
from pathlib import Path

import pytest

pytest.importorskip("torch")
pytest.importorskip("transformers")
pytest.importorskip("mcp")


def test_real_mcp_edits_and_optimizer_continuation(tmp_path):
    path = Path(__file__).resolve().parents[1] / "examples/verify_mcp_training.py"
    spec = importlib.util.spec_from_file_location("verify_mcp_training", path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    report = asyncio.run(asyncio.wait_for(module.verify(tmp_path / "verification", steps=8), timeout=90))
    result = report["cases"]["tiny"]
    assert report["passed"] and result["original_checkpoint_unchanged"]
    edits = result["edits_via_actual_stdio_mcp"]
    assert edits["identity"]["changed_parameters"] == 0
    for name in ("scale", "project_out", "rank_one"):
        assert edits[name]["changed_parameters"] > 0
        assert edits[name]["max_absolute_logit_change"] > 0
        assert edits[name]["expected_tensor_exact"] and edits[name]["actual_parameter_exact"]
    training = result["training"]
    assert training["only_edited_matrix_trainable"] and training["other_parameters_unchanged"]
    assert training["after"]["discovery"]["nll"] < training["before"]["discovery"]["nll"]
    assert training["finite_nonzero_gradients_every_step"] and training["optimizer_changed_parameters"] > 0
    assert training["mcp_edited_cells_updated_by_optimizer"] > 0
    assert training["mcp_edited_cells_nonzero_gradient_every_step"]
    assert training["checkpoint_and_logits_roundtrip_exact"] and training["trained_metrics_verified_via_mcp"]
